cmake_minimum_required(VERSION 3.24)
project(GM2627 LANGUAGES C CXX)
# Old pinned third-party projects predate CMake 4's policy minimum.
set(CMAKE_POLICY_VERSION_MINIMUM 3.5 CACHE STRING "Compatibility for pinned dependencies")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(GM_BUILD_VIEWER "Build teacher OpenGL viewer" ON)
set(GM_BUILD_SOLUTION OFF)
include(CTest)
include(cmake/Dependencies.cmake)
add_library(gm_common INTERFACE)
target_include_directories(gm_common INTERFACE common/include assignments/a01-mesh-basics/include)
target_link_libraries(gm_common INTERFACE igl::core)
set(A01 assignments/a01-mesh-basics)
foreach(impl IN ITEMS skeleton)
  if(impl STREQUAL "solution" AND NOT GM_BUILD_SOLUTION)
    continue()
  endif()
  add_library(a01_${impl} STATIC ${A01}/${impl}/a01.cpp)
  target_link_libraries(a01_${impl} PUBLIC gm_common)
  add_executable(a01_${impl}_batch ${A01}/app/batch.cpp)
  target_link_libraries(a01_${impl}_batch PRIVATE a01_${impl})
  if(GM_BUILD_VIEWER)
    add_executable(a01_${impl}_viewer ${A01}/app/viewer.cpp)
    if(APPLE)
      set_target_properties(a01_${impl}_viewer PROPERTIES MACOSX_BUNDLE TRUE
        MACOSX_BUNDLE_GUI_IDENTIFIER "edu.gm2627.a01.${impl}"
        MACOSX_BUNDLE_BUNDLE_NAME "A01 ${impl}")
    endif()
    target_link_libraries(a01_${impl}_viewer PRIVATE a01_${impl} igl::glfw igl::imgui)
    target_compile_definitions(a01_${impl}_viewer PRIVATE GM_DATA_DIR="${CMAKE_CURRENT_SOURCE_DIR}/data")
  endif()
endforeach()
if(BUILD_TESTING)
  add_executable(a01_starter_checks ${A01}/tests/starter.cpp)
  target_link_libraries(a01_starter_checks PRIVATE a01_skeleton)
  add_test(NAME starter_todo COMMAND a01_starter_checks)
  add_test(NAME starter_load COMMAND a01_skeleton_batch ${CMAKE_CURRENT_SOURCE_DIR}/data/cube.obj load)
  # Students run these as they implement the API; untouched TODOs correctly fail.
  add_executable(a01_public_checks ${A01}/tests/public.cpp)
  target_link_libraries(a01_public_checks PRIVATE a01_skeleton)
endif()
